# OreSpawn T-Rex: the Minion next to the creature (tools/smoke-orespawn.sh tools/smoke-orespawn-trex.txt)
wait 20
difficulty peaceful
gamerule doMobSpawning false
gamerule doDaylightCycle false
time set 6000
weather clear
gamerule doWeatherCycle false
hud off
kill @e[type=!player]
wait 10
summon necromancy:minion -3 -60 3 {NoAI:1b,Rotation:[0f,0f],Parts:{head:"orespawn:trex",torso:"orespawn:trex",arm_left:"orespawn:trex",arm_right:"orespawn:trex",legs:"orespawn:trex"}}
summon orespawn:trex 4 -60 3 {NoAI:1b,Rotation:[0f,0f],PersistenceRequired:1b}
tp @s 0 -58 16 180 8
wait 40
shot trex_front
tp @s -14 -57.5 4 -90 8
wait 30
shot trex_side
give @s necromancy:creature_part[necromancy:creature_part={mob:"orespawn:trex",type:"head"}]
give @s necromancy:creature_part[necromancy:creature_part={mob:"orespawn:trex",type:"torso"}]
give @s necromancy:creature_part[necromancy:creature_part={mob:"orespawn:trex",type:"legs"}]
hud on
wait 10
inventory
wait 10
shot trex_inventory
closegui
hud on
tooltip on
wait 10
shot trex_tooltip
tooltip off
report done
